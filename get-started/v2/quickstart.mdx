**<u>Sample customer support agent:</u>** Create a database, ingest a refund policy and a note about one customer, wait until both are indexed, then query. Indexing runs in the background, which is why the code waits before querying. If you already created a database, set `database` to its name and skip the create call.

Everything you ingest goes in one list called `context`. Each item carries either `text` or a `conversation`; the policy and the customer note below are both `text`.

<CodeGroup>
```python Python SDK
import json, time

database = "my_first_database"

# 1. Create a database and wait until it can accept data.
client.databases.create(database=database)
while not client.databases.status(database=database).data.infra.ready_for_ingestion:
    time.sleep(5)

# 2. Ingest the refund policy into the shared collection.
client.context.ingest(
    database=database,
    collection="company_docs",
    context=json.dumps([{
        "context_id": "refund_policy",
        "title": "Refund policy",
        "text": "Refunds are available within 30 days of purchase. Annual plans are refunded pro rata.",
    }]),
)

# 3. Save a note about one customer in that customer's collection.
client.context.ingest(
    database=database,
    collection="user_123",
    context=json.dumps([{
        "context_id": "user_123_refund_request",
        "text": "Customer asked for a refund on 2026-09-30 and was told to wait for the billing cycle to close.",
    }]),
)

# 4. Wait until both are indexed.
def wait_for(collection, id):
    while True:
        status = client.context.status(
            database=database, collection=collection, ids=[id]
        ).data.statuses[0]
        if status.indexing_status == "completed":
            return
        if status.indexing_status == "errored":
            raise RuntimeError(status.error_message)
        time.sleep(2)

wait_for("company_docs", "refund_policy")
wait_for("user_123", "user_123_refund_request")

# 5. Query the policy and the customer's note together.
results = client.query(
    database=database,
    collections=["company_docs", "user_123"],
    query="Can this customer get a refund, and what have we told them so far?",
)

for chunk in results.data.chunks:
    print(chunk.content)
```
```typescript TypeScript SDK
const database = "my_first_database";

// 1. Create a database and wait until it can accept data.
await client.databases.create({ database });
while (true) {
  const { data } = await client.databases.status({ database });
  if (data?.infra?.readyForIngestion) break;
  await new Promise((resolve) => setTimeout(resolve, 5_000));
}

// 2. Ingest the refund policy into the shared collection.
await client.context.ingest({
  database,
  collection: "company_docs",
  context: JSON.stringify([{
    context_id: "refund_policy",
    title: "Refund policy",
    text: "Refunds are available within 30 days of purchase. Annual plans are refunded pro rata.",
  }]),
});

// 3. Save a note about one customer in that customer's collection.
await client.context.ingest({
  database,
  collection: "user_123",
  context: JSON.stringify([{
    context_id: "user_123_refund_request",
    text: "Customer asked for a refund on 2026-09-30 and was told to wait for the billing cycle to close.",
  }]),
});

// 4. Wait until both are indexed.
async function waitFor(collection: string, id: string) {
  while (true) {
    const status = (await client.context.status({ database, collection, ids: [id] })).data?.statuses?.[0];
    if (!status) throw new Error("Status is missing");
    if (status.indexingStatus === "completed") return;
    if (status.indexingStatus === "errored") throw new Error(status.errorMessage ?? "Indexing failed");
    await new Promise((resolve) => setTimeout(resolve, 2_000));
  }
}

await waitFor("company_docs", "refund_policy");
await waitFor("user_123", "user_123_refund_request");

// 5. Query the policy and the customer's note together.
const results = await client.query({
  database,
  collections: ["company_docs", "user_123"],
  query: "Can this customer get a refund, and what have we told them so far?",
});

if (!results.data || typeof results.data === "string") throw new Error("Expected structured query results");
for (const chunk of results.data.chunks ?? []) {
  console.log(chunk.content);
}
```
```bash cURL
DATABASE="my_first_database"
API="https://api.hydradb.com"
AUTH=(-H "Authorization: Bearer $HYDRADB_API_KEY" -H "API-Version: 2")

# 1. Create a database and wait until it can accept data.
curl -s -X POST "$API/databases" "${AUTH[@]}" \
  -H "Content-Type: application/json" \
  -d "{\"database\":\"${DATABASE}\"}"

until curl -s "$API/databases/status?database=${DATABASE}" "${AUTH[@]}" \
  | jq -e '.data.infra.ready_for_ingestion' > /dev/null; do
  sleep 5
done

# 2. Ingest the refund policy into the shared collection.
curl -s -X POST "$API/context/ingest" "${AUTH[@]}" \
  -H "Content-Type: application/json" \
  --data-binary @- <<EOF
{
  "database": "${DATABASE}",
  "collection": "company_docs",
  "context": [
    {
      "context_id": "refund_policy",
      "title": "Refund policy",
      "text": "Refunds are available within 30 days of purchase. Annual plans are refunded pro rata."
    }
  ]
}
EOF

# 3. Save a note about one customer in that customer's collection.
curl -s -X POST "$API/context/ingest" "${AUTH[@]}" \
  -H "Content-Type: application/json" \
  --data-binary @- <<EOF
{
  "database": "${DATABASE}",
  "collection": "user_123",
  "context": [
    {
      "context_id": "user_123_refund_request",
      "text": "Customer asked for a refund on 2026-09-30 and was told to wait for the billing cycle to close."
    }
  ]
}
EOF

# 4. Wait until both are indexed.
wait_for() {
  while true; do
    STATUS=$(curl -s -G "$API/context/status" "${AUTH[@]}" \
      --data-urlencode "database=${DATABASE}" \
      --data-urlencode "collection=$1" \
      --data-urlencode "ids=$2")
    STATE=$(echo "$STATUS" | jq -r '.data.statuses[0].indexing_status')
    if [ "$STATE" = "completed" ]; then return; fi
    if [ "$STATE" = "errored" ]; then
      echo "$STATUS" | jq -r '.data.statuses[0].error_message'; exit 1
    fi
    sleep 2
  done
}
wait_for company_docs refund_policy
wait_for user_123 user_123_refund_request

# 5. Query the policy and the customer's note together.
curl -s -X POST "$API/query" "${AUTH[@]}" \
  -H "Content-Type: application/json" \
  --data-binary @- <<EOF | jq -r '.data.chunks[].content'
{
  "database": "${DATABASE}",
  "collections": ["company_docs", "user_123"],
  "query": "Can this customer get a refund, and what have we told them so far?"
}
EOF
```

## Checking the results

The query returns chunks (passages of what you ingested), not a generated answer. Each example prints the text of every chunk, which should include the policy and the customer's note:

```text
Refunds are available within 30 days of purchase. Annual plans are refunded pro rata.
Customer asked for a refund on 2026-09-30 and was told to wait for the billing cycle to close.
```

Send these chunks to your model as context, or send `llm_prompt` from the same response: the results written out as one prompt with numbered citations. See [Query](/essentials/v2/query#7-use-the-results).

HydraDB is serverless: there is nothing to provision, and the same code works at any scale, whether the context is a Markdown file or Slack messages synced by a [connector](/essentials/v2/connectors).

If your database was created before unified became the default, it is a split database and takes different fields. See [Split databases](/essentials/v2/split-databases).

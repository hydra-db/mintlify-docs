## Prerequisites

- A HydraDB API key from [dashboard.hydradb.com](https://dashboard.hydradb.com), set as `HYDRADB_API_KEY`.
- An OpenAI API key, set as `OPENAI_API_KEY`. HydraDB returns context, not answers, so any model works; swap the `answer` function below for your own.
- Python 3.10 or later, with the SDKs: `pip install "hydradb-sdk>=2.1.7,<3" openai`.

Save this as `hydra.py`. Every cookbook imports it. It holds what every HydraDB application does around its own logic: wait for the database, connect an app, check what an ingest accepted, wait for indexing, and pass query results to a model.

```python hydra.py
import os
import time

from hydra_db import HydraDB
from hydra_db.errors import ConflictError
from hydra_db.helpers import build_string
from openai import OpenAI

client = HydraDB(token=os.environ["HYDRADB_API_KEY"])
llm = OpenAI()


def create_database(database, schema=None):
    """Create the database if it is new, then wait until it accepts context."""
    try:
        if schema:
            client.databases.create(database=database, database_metadata_schema=schema)
        else:
            client.databases.create(database=database)
    except ConflictError:
        pass  # The database already exists.
    while not client.databases.status(database=database).data.infra.ready_for_ingestion:
        time.sleep(5)


# Providers that take an account scope. Without one, two accounts whose
# sources share an ID overwrite each other.
SCOPED = {"slack", "github", "linear", "notion"}


def connect(provider, account, database, collection, credentials,
            acl=None, lookback_days=30, timeout=1800):
    """Connect one app account, sync what its credentials can reach, and wait for the first sync.

    account names the account the credentials belong to: a Slack workspace ID,
    a mailbox address, a Jira site. Each account gets its own connector.
    Pass acl for a provider that does not capture permissions, such as Gmail or
    Notion. Without it, everything that provider syncs is readable by every caller.
    connect refuses such a provider when acl is left out, and stops if a provider's
    permissions could not be read. An empty list lets nobody read the content.
    The credentials are used when the connector
    is created; to rotate them later, call client.connectors.update.
    """
    if acl is None:  # Then only the provider's own permissions restrict this content.
        catalog = client.list_providers().providers or []
        if not any(p.provider == provider and getattr(p, "acl_supported", False) for p in catalog):
            raise ValueError(f"{provider} captures no permissions: pass acl to say who may read it")
    name = f"cookbook-{provider}-{collection}-{account}"
    connectors = client.connectors.list(provider=provider).connectors or []
    existing = [c for c in connectors if c.name == name and c.database == database]
    if existing:  # This account was connected on an earlier run.
        connector_id = existing[0].connector_id
    else:
        scope = {"provider_account_scope": account} if provider in SCOPED else {}
        connector_id = client.connectors.create(
            provider=provider,
            name=name,
            database=database,
            collection=collection,
            credentials=credentials,
            **scope,
        ).connector_id

    if client.connectors.status(connector_id).lifecycle == "pending_setup":
        resources, cursor = [], None
        while True:  # Discover returns up to 100 resources per page.
            page = client.connectors.discover(connector_id, cursor=cursor, limit=100)
            resources += page.resources or []
            if not page.has_more:
                break
            cursor = page.next_cursor
        if not resources:
            raise RuntimeError(f"The {provider} credentials can reach nothing to sync")
        rule = {} if acl is None else {"acl": acl}
        client.connectors.configure(
            connector_id,
            lookback_days=lookback_days,
            resources=[
                {"resource_id": r.id, "resource_type": r.resource_type, "name": r.name, **rule}
                for r in resources
            ],
        )
    elif acl is not None:  # Already configured: set the rule asked for now on every resource.
        for resource in client.connectors.list_resources(connector_id).resources or []:
            client.connectors.update_resource_acl(
                connector_id, resource.resource_id, request={"acl": acl}
            )

    deadline = time.time() + timeout
    while time.time() < deadline:
        status = client.connectors.status(connector_id)
        if status.status in ("failed", "paused", "capped") or status.lifecycle == "reconnect":
            raise RuntimeError(f"The {provider} connector is {status.status}: {status.error}")
        failed = [r for r in status.resources or [] if r.status == "failed"]
        detail = "; ".join(f"{r.display_name or r.resource_id}: {r.message}" for r in failed)
        if any(r.retryable is False for r in failed):  # Waiting will not fix these.
            raise RuntimeError(f"The {provider} connector cannot sync {detail}")
        if status.lifecycle == "active":
            unread = [
                r.display_name or r.resource_id
                for r in status.resources or []
                if r.acl_warning or r.page_acl_warning
            ]
            if acl is None and unread:  # Synced without permissions: readable by everyone.
                raise RuntimeError(
                    f"The {provider} connector could not read the permissions of "
                    + ", ".join(unread)
                    + ". Pass acl to restrict them."
                )
            if status.status == "healthy":
                return connector_id
            raise RuntimeError(
                f"The {provider} connector is {status.status}, so its content is incomplete: "
                + (detail or str(status.error))
            )
        time.sleep(10)
    raise TimeoutError(
        f"The {provider} connector synced nothing within {timeout} seconds"
        + (f" ({detail})" if detail else "")
    )


def source_ids(result):
    """Return the IDs an ingest accepted. Raise if it rejected any item."""
    for item in result.data.results:
        if item.error_code:
            raise RuntimeError(f"{item.filename or item.id}: {item.error_code} {item.error}")
    return [item.id for item in result.data.results]


def wait_indexed(database, collection, ids):
    """Block until every source has finished indexing. Indexing runs in the background."""
    while True:
        statuses = client.context.status(
            database=database, collection=collection, ids=ids
        ).data.statuses
        for status in statuses:
            if status.indexing_status == "errored":
                raise RuntimeError(f"{status.error_code}: {status.error_message}")
        if all(s.indexing_status == "completed" for s in statuses):
            return
        time.sleep(2)


def answer(question, result, instructions=""):
    """Send a query result to the model and return its answer."""
    completion = llm.chat.completions.create(
        model="gpt-4o",
        messages=[
            {
                "role": "system",
                "content": "Answer using only the context provided. If the answer "
                "is not in the context, say you don't know. " + instructions,
            },
            {
                "role": "user",
                "content": f"Context:\n{build_string(result)}\n\nQuestion: {question}",
            },
        ],
    )
    return completion.choices[0].message.content
```

## Prerequisites

- A HydraDB API key from [dashboard.hydradb.com](https://dashboard.hydradb.com), set as `HYDRADB_API_KEY`.
- An OpenAI API key, set as `OPENAI_API_KEY`. HydraDB returns context, not answers, so any model works; swap the `answer` function below for your own.
- Python 3.10 or later, with the SDKs: `pip install "hydradb-sdk>=2.1.7,<3" openai`.

Save this as `hydra.py`. Every cookbook imports it. It holds what every HydraDB application does around its own logic: wait for the database, connect an app, check what an ingest accepted, wait for indexing, and pass query results to a model.

```python hydra.py
import os
import time

from hydra_db import HydraDB
from hydra_db.errors import ConflictError
from hydra_db.helpers import build_string
from openai import OpenAI

client = HydraDB(token=os.environ["HYDRADB_API_KEY"])
llm = OpenAI()


def create_database(database, schema=None):
    """Create the database if it is new, then wait until it accepts context."""
    try:
        if schema:
            client.databases.create(database=database, database_metadata_schema=schema)
        else:
            client.databases.create(database=database)
    except ConflictError:
        pass  # The database already exists.
    while not client.databases.status(database=database).data.infra.ready_for_ingestion:
        time.sleep(5)


def connect(provider, database, collection, credentials, account_scope=None):
    """Connect one app account, sync what its credentials can reach, and wait for the first sync."""
    existing = [
        c for c in client.connectors.list(provider=provider).connectors or []
        if c.database == database and c.collection == collection
    ]
    if existing:  # Already connected on an earlier run.
        connector_id = existing[0].connector_id
    else:
        scope = {"provider_account_scope": account_scope} if account_scope else {}
        connector_id = client.connectors.create(
            provider=provider,
            database=database,
            collection=collection,
            credentials=credentials,
            **scope,
        ).connector_id
        found = client.connectors.discover(connector_id)
        client.connectors.configure(
            connector_id,
            resources=[
                {"resource_id": r.id, "resource_type": r.resource_type, "name": r.name}
                for r in found.resources
            ],
        )
    while True:
        status = client.connectors.status(connector_id)
        if status.lifecycle == "active":
            return connector_id
        if status.status == "failed":
            raise RuntimeError(f"{provider} connector failed: {status.error}")
        time.sleep(10)


def source_ids(result):
    """Return the IDs an ingest accepted. Raise if it rejected any item."""
    for item in result.data.results:
        if item.error_code:
            raise RuntimeError(f"{item.filename or item.id}: {item.error_code} {item.error}")
    return [item.id for item in result.data.results]


def wait_indexed(database, collection, ids):
    """Block until every source has finished indexing. Indexing runs in the background."""
    while True:
        statuses = client.context.status(
            database=database, collection=collection, ids=ids
        ).data.statuses
        for status in statuses:
            if status.indexing_status == "errored":
                raise RuntimeError(f"{status.error_code}: {status.error_message}")
        if all(s.indexing_status == "completed" for s in statuses):
            return
        time.sleep(2)


def answer(question, result, instructions=""):
    """Send a query result to the model and return its answer."""
    completion = llm.chat.completions.create(
        model="gpt-4o",
        messages=[
            {
                "role": "system",
                "content": "Answer using only the context provided. If the answer "
                "is not in the context, say you don't know. " + instructions,
            },
            {
                "role": "user",
                "content": f"Context:\n{build_string(result)}\n\nQuestion: {question}",
            },
        ],
    )
    return completion.choices[0].message.content
```
